// ============================================================ // ENVIRONMENT MAP — OPTIMIZOVAN BACKGROUND // ============================================================ const mapTextureCanvas = document.createElement('canvas'); mapTextureCanvas.width = V_WIDTH; mapTextureCanvas.height = V_HEIGHT; const mctx = mapTextureCanvas.getContext('2d', { alpha: false }); // Ovde možeš kasnije staviti svoju JPG/PNG mapu. // Ako je fajl npr. "environment.jpg", mora biti u istom folderu // kao HTML fajl. const environmentImage = new Image(); environmentImage.src = "environment.jpg"; let environmentLoaded = false; environmentImage.onload = () => { environmentLoaded = true; renderEnvironmentMap(); }; environmentImage.onerror = () => { console.warn("[ENVIRONMENT] JPG nije pronađen — koristi se ugrađena mapa."); renderFallbackMap(); }; // ============================================================ // RENDER ENVIRONMENT // ============================================================ function renderEnvironmentMap() { // Crna osnova mctx.fillStyle = "#01040a"; mctx.fillRect(0, 0, V_WIDTH, V_HEIGHT); // Cover scaling — nema deformacije slike const imageRatio = environmentImage.width / environmentImage.height; const canvasRatio = V_WIDTH / V_HEIGHT; let drawWidth; let drawHeight; let offsetX; let offsetY; if (imageRatio > canvasRatio) { drawHeight = V_HEIGHT; drawWidth = V_HEIGHT * imageRatio; offsetX = (V_WIDTH - drawWidth) / 2; offsetY = 0; } else { drawWidth = V_WIDTH; drawHeight = V_WIDTH / imageRatio; offsetX = 0; offsetY = (V_HEIGHT - drawHeight) / 2; } mctx.drawImage( environmentImage, offsetX, offsetY, drawWidth, drawHeight ); // Tamni strateški overlay mctx.fillStyle = "rgba(0, 5, 15, 0.30)"; mctx.fillRect(0, 0, V_WIDTH, V_HEIGHT); // GRID mctx.strokeStyle = "rgba(0,255,204,0.08)"; mctx.lineWidth = 1; const gridSize = 60; for (let x = 0; x <= V_WIDTH; x += gridSize) { mctx.beginPath(); mctx.moveTo(x, 0); mctx.lineTo(x, V_HEIGHT); mctx.stroke(); } for (let y = 0; y <= V_HEIGHT; y += gridSize) { mctx.beginPath(); mctx.moveTo(0, y); mctx.lineTo(V_WIDTH, y); mctx.stroke(); } // Centralna zona const cx = V_WIDTH / 2; const cy = V_HEIGHT / 2; const centralGlow = mctx.createRadialGradient( cx, cy, 20, cx, cy, 420 ); centralGlow.addColorStop( 0, "rgba(0,255,204,0.14)" ); centralGlow.addColorStop( 1, "rgba(0,255,204,0)" ); mctx.fillStyle = centralGlow; mctx.beginPath(); mctx.arc(cx, cy, 420, 0, Math.PI * 2); mctx.fill(); // Vignette const vignette = mctx.createRadialGradient( cx, cy, 250, cx, cy, Math.max(V_WIDTH, V_HEIGHT) * 0.75 ); vignette.addColorStop( 0, "rgba(0,0,0,0)" ); vignette.addColorStop( 1, "rgba(0,0,0,0.65)" ); mctx.fillStyle = vignette; mctx.fillRect( 0, 0, V_WIDTH, V_HEIGHT ); console.log( "[ENVIRONMENT] Environment mapa učitana." ); } // ============================================================ // FALLBACK MAP // Ako JPG ne postoji, igra i dalje radi. // ============================================================ function renderFallbackMap() { mctx.fillStyle = "#01040a"; mctx.fillRect( 0, 0, V_WIDTH, V_HEIGHT ); const cx = V_WIDTH / 2; const cy = V_HEIGHT / 2; // Energetske zone for (let i = 0; i < 10; i++) { const x = Math.random() * V_WIDTH; const y = Math.random() * V_HEIGHT; const radius = 120 + Math.random() * 300; const gradient = mctx.createRadialGradient( x, y, 5, x, y, radius ); gradient.addColorStop( 0, "rgba(0,255,204,0.16)" ); gradient.addColorStop( 0.5, "rgba(0,100,255,0.06)" ); gradient.addColorStop( 1, "rgba(0,0,0,0)" ); mctx.fillStyle = gradient; mctx.beginPath(); mctx.arc( x, y, radius, 0, Math.PI * 2 ); mctx.fill(); } // Zvezde for (let i = 0; i < 600; i++) { const x = Math.random() * V_WIDTH; const y = Math.random() * V_HEIGHT; const size = Math.random() * 1.6; mctx.fillStyle = `rgba(255,255,255,${0.2 + Math.random() * 0.8})`; mctx.beginPath(); mctx.arc( x, y, size, 0, Math.PI * 2 ); mctx.fill(); } // GRID mctx.strokeStyle = "rgba(0,255,204,0.07)"; for ( let x = 0; x < V_WIDTH; x += 60 ) { mctx.beginPath(); mctx.moveTo(x, 0); mctx.lineTo(x, V_HEIGHT); mctx.stroke(); } for ( let y = 0; y < V_HEIGHT; y += 60 ) { mctx.beginPath(); mctx.moveTo(0, y); mctx.lineTo(V_WIDTH, y); mctx.stroke(); } console.log( "[ENVIRONMENT] Fallback mapa aktivna." ); } // ============================================================ // DRAW BACKGROUND // ============================================================ function drawEnvironment() { ctx.drawImage( mapTextureCanvas, 0, 0 ); }
