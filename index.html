const canvas = document.getElementById("warCanvas"); const ctx = canvas.getContext("2d", { alpha: false, desynchronized: true }); const canvasArea = document.getElementById("canvasArea"); /* ============================================================ HONOR 90 — ADAPTIVE MAX RESOLUTION ============================================================ Honor 90 fizička rezolucija: 2664 × 1200 Ne koristimo fiksnih 3840×2160. Canvas se automatski prilagođava stvarnom ekranu. devicePixelRatio omogućava maksimalnu oštrinu. */ const HONOR90_WIDTH = 2664; const HONOR90_HEIGHT = 1200; let WORLD_WIDTH = HONOR90_WIDTH; let WORLD_HEIGHT = HONOR90_HEIGHT; let renderScale = 1; function getMaxResolution() { const dpr = window.devicePixelRatio || 1; /* Na Honor 90 browser viewport može biti manji od fizičke rezolucije zbog navigacije/status bara. Uzimamo stvarnu dostupnu veličinu. */ const screenW = Math.max( window.innerWidth, window.screen.width ); const screenH = Math.max( window.innerHeight, window.screen.height ); /* Ako je telefon u portrait modu, širina/visina se automatski menjaju. */ const physicalW = Math.max( screenW * dpr, 1 ); const physicalH = Math.max( screenH * dpr, 1 ); /* Ne dozvoljavamo ekstremno veliki canvas koji bi ubio FPS na telefonu. */ const MAX_PIXELS = 2664 * 1200; let w = physicalW; let h = physicalH; const pixels = w * h; if (pixels > MAX_PIXELS) { const scale = Math.sqrt( MAX_PIXELS / pixels ); w *= scale; h *= scale; } return { width: Math.floor(w), height: Math.floor(h) }; } /* ============================================================ CANVAS RESIZE ============================================================ */ function resizeCanvas() { const areaW = canvasArea.clientWidth; const areaH = canvasArea.clientHeight; if (areaW <= 0 || areaH <= 0) return; /* Čuvamo odnos 16:9. */ const aspect = 16 / 9; let displayW = areaW; let displayH = displayW / aspect; if (displayH > areaH) { displayH = areaH; displayW = displayH * aspect; } /* Maksimalni fizički render. */ const maxRes = getMaxResolution(); WORLD_WIDTH = maxRes.width; WORLD_HEIGHT = maxRes.height; /* Canvas fizička rezolucija. */ canvas.width = WORLD_WIDTH; canvas.height = WORLD_HEIGHT; /* CSS veličina. */ canvas.style.width = Math.floor(displayW) + "px"; canvas.style.height = Math.floor(displayH) + "px"; canvas.style.position = "absolute"; canvas.style.left = "50%"; canvas.style.top = "50%"; canvas.style.transform = "translate(-50%, -50%)"; /* Informacija za debug. */ console.log( "HONOR 90 RENDER:", WORLD_WIDTH, "×", WORLD_HEIGHT, "DPR:", window.devicePixelRatio ); } /* ============================================================ ROTACIJA TELEFONA ============================================================ */ window.addEventListener( "resize", resizeCanvas ); window.addEventListener( "orientationchange", () => { setTimeout(() => { resizeCanvas(); }, 300); } ); /* ============================================================ POČETNO PODEŠAVANJE ============================================================ */ resizeCanvas();
