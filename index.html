/* RANDOM GENERATOR 50 ITEMA SA 6 STATOVA (250G - 5000G) */ const namePrefixes = [ "Runic", "Void", "Shadow", "Celestial", "Infernal", "Glacial", "Arcane", "Titan's", "Sovereign", "Storm", "Phantasmal", "Abyssal", "Ethereal", "Vigilant", "Mythic", "Ancient" ]; const nameTypes = [ "Blade", "Amulet", "Aegis", "Grasp", "Tome", "Crown", "Scepter", "Visage", "Heart", "Cuirass", "Orb", "Relic", "Edge", "Plate", "Shard", "Ring" ]; const nameSuffixes = [ "of the Phoenix", "of the Abyss", "of Eternity", "of the Storm", "of Ruin", "of the Titan", "of Shadows", "of Redemption", "of the Void", "of Might" ]; function generate50Items(){ const list = []; const categories = ["AD", "AP", "DEF"]; for(let i = 0; i < 50; i++){ // Skaliranje cene od 250G do 5000G const priceRatio = i / 49; // 0 na startu, 1 na kraju const price = Math.round((250 + priceRatio * 4750) / 10) * 10; // Zaokruženo na 10G const cat = categories[i % 3]; // Generisanje nasumičnog imena const pref = namePrefixes[Math.floor(Math.random() * namePrefixes.length)]; const type = nameTypes[Math.floor(Math.random() * nameTypes.length)]; const suff = nameSuffixes[Math.floor(Math.random() * nameSuffixes.length)]; const itemName = `${pref} ${type} ${suff}`; // Množilac u zavisnosti od cene (skuplji item = drastično jači statovi) const power = price / 250; // Generisanje svih 6 traženih statova const item = { id: "item_" + (i + 1), name: itemName, cat: cat, price: price, // Merni statovi hp: Math.floor((Math.random() * 80 + 40) * power), // Health speed: parseFloat((Math.random() * 2 + 1 * (power * 0.5)).toFixed(1)), // Brzina att: Math.floor((cat === "AD" ? 15 : 5) * power + Math.random() * 10), // Napad def: Math.floor((cat === "DEF" ? 10 : 3) * power + Math.random() * 5), // Odbrana heal: Math.floor((Math.random() * 10 + 5) * power), // Hilovanje (HP/s ili Lifesteal) manaRegen: parseFloat((Math.random() * 1.5 + 0.5 * power).toFixed(1)), // Vraćanje mane/s desc: `+${Math.floor(power * 20)} Power Rating` }; list.push(item); } // Sortiramo po ceni od 250G do 5000G return list.sort((a, b) => a.price - b.price); }
